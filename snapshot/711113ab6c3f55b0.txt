    function renderPageData(page) {
        var title = $('#txtTitle').val();
        var orderBy = $('#ddOrderBy').val();

        if (page && title && orderBy) {
            location.href = '/News-List?' + 'title=' + title + '&orderby=' + orderBy + '&page=' + page;
        } else if (page && title) {
            location.href = '/News-List?' + 'title=' + title + '&page=' + page;
        } else if (page && orderBy) {
            location.href = '/News-List?' + 'orderby=' + orderBy + '&page=' + page;
        } else if (page) {
            location.href = '/News-List?' + 'page=' + page;
        }
    }

    $(document).ready(function () {

        $("#ddOrderBy").val("");
        $("#txtTitle").val("");


        $("#ddOrderBy").change(function () {
            var orderBy = $(this).val();
            var title = $('#txtTitle').val();

            formURLAndRedirect(title, orderBy);
        });

        $('#search').click(function () {
            var title = $('#txtTitle').val();
            var orderBy = $('#ddOrderBy').val();

            formURLAndRedirect(title, orderBy)
        });

        function formURLAndRedirect(title, orderBy) {
            if (title && orderBy) {
                location.href = '/News-List?' + 'title=' + title + '&orderby=' + orderBy;
            } else if (title) {
                location.href = '/News-List?' + 'title=' + title;
            } else if (orderBy) {
                location.href = '/News-List?' + 'orderby=' + orderBy;
            } else {
                location.href = '/News-List';
            }
        }
    });
</script>
